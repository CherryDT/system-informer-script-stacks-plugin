cmake_minimum_required(VERSION 3.24)
project(ScriptStacks VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the Visual Studio x64 toolchain; run scripts/build.ps1.")
endif()
set(SYSTEM_INFORMER_SDK "${CMAKE_SOURCE_DIR}/.deps/sdk" CACHE PATH "Pinned System Informer SDK")
file(GLOB PLUGIN_SOURCES CONFIGURE_DEPENDS "plugin/*.cpp" "plugin/*.c" "plugin/*.rc")
add_library(ScriptStacks SHARED ${PLUGIN_SOURCES})
target_compile_features(ScriptStacks PRIVATE cxx_std_17)
set_property(TARGET ScriptStacks PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(ScriptStacks PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(ScriptStacks PRIVATE /W4 /Zi /utf-8 $<$<COMPILE_LANGUAGE:C,CXX>:/guard:cf> $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
target_include_directories(ScriptStacks SYSTEM PRIVATE "${SYSTEM_INFORMER_SDK}/include" "${CMAKE_SOURCE_DIR}/vendor")
target_link_libraries(ScriptStacks PRIVATE "${SYSTEM_INFORMER_SDK}/lib/SystemInformer.lib"
    ntdll comctl32 comdlg32 user32 gdi32 advapi32 shell32 shlwapi uxtheme ole32 version psapi)
target_link_options(ScriptStacks PRIVATE /DEBUG:FULL /OPT:REF /OPT:ICF /guard:cf)
set_target_properties(ScriptStacks PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist")
